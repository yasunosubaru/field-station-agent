# ===== 文件：src/fieldstation/agents/approval.py =====
"""人工审批门：高风险动作必须有人批过才执行。

spec §2 与 README 的「规则裁决，模型归因」是同一条约束的两面：**模型只负责归因，
不确定性规则负责裁决，而高风险动作必须留人工审批门**。一次错误的停机建议可能
造成不可逆的数据损失，所以「不动作」必须始终是可选项。

本模块是那条约束的落地。三层结构，每层都能单独换：

* ``ApprovalRequest`` —— 要批什么（agent、动作、理由、置信度、风险级）
* ``Approver`` —— 谁来批。现在是脚本化的 ``ApproveAll`` / ``RejectAll``，
  M3 之后换成真实的人机界面，**本文件不需要改**
* ``ApprovalGate`` —— 什么情况下需要批（风险级命中，或置信度低于阈值）

设计取舍：

* **没有审批者时不给默认放行。** ``ApprovalGate`` 必须显式拿到一个 approver；
  没配 approver 就不接审批请求，宁可让调用方在启动时就失败。默认放行等于
  把安全约束交给人有没有记得配。
* **审批记录不含时间戳或随机量。** 它要进 trace，而 trace 的同输入两跑
  逐字节相同是回放式评测的地基。时间戳放 ``store.product.env_fingerprint()``
  那类旁路产物里。
"""
from __future__ import annotations

from dataclasses import dataclass
from typing import Protocol, runtime_checkable

#: 默认必须人工审批的风险级
DEFAULT_RISK_LEVELS = ("high",)

#: 默认置信度阈值：低于它就算模型自己没把握，要人看一眼
DEFAULT_CONFIDENCE_THRESHOLD = 0.9

#: 允许的风险级取值。写死的白名单，拼错一个字母就等于静默关掉审批。
RISK_LEVELS = ("low", "medium", "high")


@dataclass

    """一次待批的动作。``confidence`` 取 0-1，模型没给就默认 1.0。"""

    agent: str
    action: str
    reason: str
    confidence: float = 1.0
    risk: str = "low"

@dataclass
class ApprovalRecord:
    """审批结论。字段刻意铺平而不是嵌一个 request 对象 —— 要直接进 trace，
    嵌套结构会让 JSON 化与比对都变麻烦。"""

    verdict: str          # approved / rejected / pending
    agent: str
    action: str
    reason: str
    confidence: float
    risk: str
    by: str


@runtime_checkable
class Approver(Protocol):
    def decide(self, request: ApprovalRequest) -> ApprovalRecord:
        ...


class ApproveAll:
    by = "approve_all"

    def decide(self, request: ApprovalRequest) -> ApprovalRecord:
        return self._record(request, "approved")

    @staticmethod
    def _record(request: ApprovalRequest, verdict: str) -> ApprovalRecord:
        return ApprovalRecord(
            verdict=verdict, agent=request.agent, action=request.action,
            reason=request.reason, confidence=request.confidence,
            risk=request.risk, by="approve_all")


class RejectAll:
    by = "reject_all"

    def decide(self, request: ApprovalRequest) -> ApprovalRecord:
        return ApprovalRecord(
            verdict="rejected", agent=request.agent, action=request.action,
            reason=request.reason, confidence=request.confidence,
            risk=request.risk, by="reject_all")


class ApprovalGate:
    """决定「这个动作要不要人批」，并拿到结论。

    ``needs_approval`` 为假时 ``decide`` 直接返回 ``approved`` 且 ``by="auto"``
    —— 不给低风险动作增加打扰，一次纯统计查询不该走一遍审批。

------------------------------------------------------------------------------
野外台站智能体运行与评估系统 V1.0                                          第 3 页 / 共 169 页
    """

    def __init__(
        self,
        approver: Approver,
        confidence_threshold: float = DEFAULT_CONFIDENCE_THRESHOLD,
        risk_needing_approval: tuple[str, ...] = DEFAULT_RISK_LEVELS,
    ) -> None:
        bad = [r for r in risk_needing_approval if r not in RISK_LEVELS]
        if bad:
            raise ValueError(f"未知风险级：{bad}（只认 {RISK_LEVELS}）")
        self.approver = approver
        self.confidence_threshold = confidence_threshold
        self.risk_needing_approval = tuple(risk_needing_approval)

    def needs_approval(self, request: ApprovalRequest) -> bool:
        if request.risk in self.risk_needing_approval:
            return True
        return request.confidence < self.confidence_threshold

    def decide(self, request: ApprovalRequest) -> ApprovalRecord:
        if not self.needs_approval(request):
            return ApprovalRecord(
                verdict="approved", agent=request.agent, action=request.action,
                reason=request.reason, confidence=request.confidence,
                risk=request.risk, by="auto")
        return self.approver.decide(request)

"""共享黑板：台站 / 告警 / 工单上下文的受控读写（spec §4「上下文管理」）。

M5-1 把「agent 之间的数据通道」钉成消息总线；本模块补上另一半 —— **跨阶段的
上下文**：台站健康度、告警、工单这类「多个人要看、按需更新」的状态。spec §4 对
它的要求是两条：读写受控、并发访问语义明确。这里把它们变成可测的代码，语义说明
另见 ``docs/operations/blackboard.md``。

## 数据形状

* 键：``<命名空间>/<路径...>``。命名空间固定三类 —— ``station``（台站）、
  ``alert``（告警）、``order``（工单），与 spec §4 的括号逐字对应；路径段非空。
  例：``station/BADG``、``station/BADG/health``、``alert/A-0001``、``order/WO-0001``。
* 值：**dict**，且必须可 JSON 序列化（非 JSON 类型与 NaN/Inf 一律拒收 —— 与总线的
  序列化纪律同一条）。**整值替换，不提供就地修改**：想改就给一个新 dict。没有
  「就地改一半」这个口子，读者就不会看到写者改到一半的半个状态。
* 每个键带只增不减的版本号（创建从 1 开始，每次写 +1；**删除后重建也不回退**，
  从上一版本继续加，防止删除前拿到的 CAS token 在重建后命中新条目）与最近写入者
  ``by``。

## 并发语义（tests/test_m5_blackboard.py 用真线程钉死）


------------------------------------------------------------------------------

* **进程内、线程安全**：所有操作在一把 ``threading.RLock`` 下原子完成；值进出都
  深拷贝 —— 读到的对象与黑板内部不共享引用。
* **写分三种，默认路径是 CAS**：

  - ``create(key, value)``：键必须不存在，否则 ``BoardConflict``；
  - ``update(key, value, expect_version=n)``：版本不符 ``BoardConflict``；
  - ``upsert(key, value)``：无条件整值替换（显式 last-writer-wins），版本仍自增；
  - ``delete(key, expect_version=n)``：版本校验同 ``update``。

  静默覆盖会丢更新：两个写者各按 ``version=3`` 做决定、后写的赢，前一个的判断
  连同依据一起消失且没有痕迹。所以默认路径要带 ``expect_version``；无条件路径
  必须显式写 ``upsert``。
* **单键原子、多键不原子**：跨键一致性不在这里保证 —— 要喂给下游的走消息，
  要在黑板里保持一致的聚合成一个键。
* **变更事件**（``on_event``，可选）在锁内按**提交顺序**同步回调；**读不发事件**
  （读的留痕由工具层的 ``tool_call`` 覆盖）。回调要轻量：它挡着所有写者；
  同线程回调可重入（RLock）。
* 锁内不跑用户代码之外的东西：没有后台线程、没有定时器、没有中间件。

## 受控访问

``board.scope(reads=..., writes=..., actor=...)`` 生成 ``BoardView``。范围是一组
**前缀模式**：``"station"``（该命名空间全部）、``"station/BADG"``（含其子键）、
``"*"``（全部）。键等于模式、或以 ``模式 + "/"`` 开头才算命中 —— 所以
``station/BADG`` 不会命中 ``station/BADGX``。越权访问抛 ``BoardPermissionError``；
工具层（``tools/board_tools.py``）再把它翻成 ``ToolResult(ok=False, ...)`` ——
库层用异常、agent 层用失败结果，两侧的既有约定各自成立。

``pipeline.Pipeline`` 按每个阶段的 ``board_reads`` / ``board_writes`` 声明构造视图
并注入黑板工具：**声明是唯一的能力与范围来源**，阶段不声明，一个方向也碰不到。

本模块只用标准库。

def check_key(key: object) -> str:
    """校验**条目键**的形状（命名空间 + 非空路径段），返回键本身。"""
    if not isinstance(key, str) or not key.strip():
        raise ValueError("黑板键必须是非空字符串")
    parts = key.split("/")
    if any(not part.strip() for part in parts):
        raise ValueError(f"黑板键 {key!r} 含空路径段")
    if parts[0] not in NAMESPACES:
        raise ValueError(
            f"黑板键 {key!r} 的命名空间必须在 {list(NAMESPACES)} 内")
    if len(parts) < 2:
        raise ValueError(
            f"黑板键 {key!r} 缺少路径段（命名空间下要有一个条目名，如 station/BADG）")
    return key

def check_prefix(prefix: object) -> str:
    """校验**前缀**（命名空间或条目键）：``snapshot`` 的子树过滤与范围模式用。"""
    if not isinstance(prefix, str) or not prefix.strip():
        raise ValueError("黑板前缀必须是非空字符串")
    parts = prefix.split("/")
    if any(not part.strip() for part in parts):
        raise ValueError(f"黑板前缀 {prefix!r} 含空路径段")
    if parts[0] not in NAMESPACES:
        raise ValueError(
            f"黑板前缀 {prefix!r} 的命名空间必须在 {list(NAMESPACES)} 内")
    return prefix


def check_value(value: object) -> None:
    """校验值的形状：必须是可 JSON 序列化的 dict（拒 NaN/Inf）。"""
    if not isinstance(value, dict):
        raise ValueError("黑板值必须是 dict（整值替换，不做就地修改）")
    try:
        json.dumps(value, sort_keys=True, ensure_ascii=False, allow_nan=False)
    except (TypeError, ValueError) as exc:
        raise ValueError(
            f"黑板值必须可 JSON 序列化（拒 NaN/Inf）：{exc}") from exc


def check_patterns(patterns: Iterable[str]) -> tuple[str, ...]:
    """校验范围模式（``*``、命名空间或条目键前缀），去重且保持顺序。"""
    out: list[str] = []
    for pattern in patterns:
        if not isinstance(pattern, str):
            raise ValueError(f"范围模式必须是字符串：{pattern!r}")
        if pattern != "*":
            check_prefix(pattern)
        if pattern not in out:
            out.append(pattern)
    return tuple(out)


def _matches(key: str, patterns: tuple[str, ...]) -> bool:
    for pattern in patterns:
        if pattern == "*" or key == pattern or key.startswith(pattern + "/"):
            return True
    return False


def _copy_entry(entry: BoardEntry) -> BoardEntry:
    return BoardEntry(key=entry.key, value=copy.deepcopy(entry.value),
                      version=entry.version, by=entry.by)

    """进程内共享上下文：受控读写、CAS 写、真线程安全。

    ``on_event`` 收到的是变更事件（``op`` ∈ create/update/upsert/delete，
    带 ``key`` / ``version`` / ``by``），锁内按提交顺序同步调用。
    """

    def __init__(self, *, on_event: Callable[[dict], None] | None = None) -> None:
        self._lock = threading.RLock()
        self._entries: dict[str, BoardEntry] = {}
        #: 每个键见过的最大版本。删除不清理 —— 删除后重建的版本继续往上加，
        #: 否则删除前拿到的 CAS token 会在重建后命中新条目（ABA）。
        self._versions: dict[str, int] = {}
        self._on_event = on_event

    # ------------------------------------------------------------------ 读
    def read(self, key: str) -> BoardEntry | None:
        """读一个键；不存在返回 ``None``，返回值是深拷贝。"""
        key = check_key(key)
        with self._lock:
            entry = self._entries.get(key)
            return None if entry is None else _copy_entry(entry)

    def snapshot(self, prefix: str | None = None) -> dict[str, BoardEntry]:
        """一致性快照（锁内一次性深拷贝）。``prefix`` 给命名空间或键前缀即只取该子树。"""
        if prefix is not None:
            prefix = check_prefix(prefix)
        with self._lock:
            return {
                key: _copy_entry(entry)
                for key, entry in sorted(self._entries.items())
                if prefix is None or key == prefix or key.startswith(prefix + "/")
            }

    def keys(self) -> tuple[str, ...]:
        with self._lock:
            return tuple(sorted(self._entries))

class BoardView:
    """黑板的一个受控视图：只能碰 ``reads`` / ``writes`` 范围内的键。

    范围语义与 ``Blackboard.scope`` 一致（前缀模式，``*`` 全通）。
    越权一律抛 ``BoardPermissionError``，不静默返回空。
    """

    def __init__(self, board: Blackboard, reads: tuple[str, ...],
                 writes: tuple[str, ...], *, actor: str = "") -> None:
        self._board = board
        self.reads = reads
        self.writes = writes
        self.actor = actor

    def _ensure(self, key: object, patterns: tuple[str, ...], op: str) -> str:
        key = check_key(key)
        if not _matches(key, patterns):
            raise BoardPermissionError(key, op, patterns)
        return key

    def scope_dict(self) -> dict:
        """范围的可序列化描述（进工具描述/trace 用）。"""
        return {"reads": list(self.reads), "writes": list(self.writes),
                "actor": self.actor}

    # 读
    def read(self, key: str) -> BoardEntry | None:
        key = self._ensure(key, self.reads, "read")
        return self._board.read(key)

    def snapshot(self, prefix: str | None = None) -> dict[str, BoardEntry]:
        """只含读范围内的键；``prefix`` 另按前缀过滤（可落在范围外，则得空）。"""
        entries = self._board.snapshot(prefix)
        return {key: entry for key, entry in entries.items()
                if _matches(key, self.reads)}

    def keys(self) -> tuple[str, ...]:
        return tuple(key for key in self._board.keys()
                     if _matches(key, self.reads))

"""进程内消息总线：agent 之间的唯一通信通道，消息全量落 trace（M5-1，issue #60）。

spec §4 的「消息路由」与 §8 的 YAGNI 放在一起得出的形状：五类 Agent 跑在同一条
进程内链路上，**不引任何消息队列中间件**。本模块只做四件事：

* 顺序号消息信封（``Message``）—— 谁、对哪个主题、说了什么、引了哪些上游消息；
* 订阅与投递 —— 收件箱按「发布顺序」排列，没有优先级、没有重投、没有超时；
* 全量留痕 —— 每条消息以 ``message`` 事件写进 trace，**带 payload 本体**；
* 序列化纪律 —— payload 必须是可 JSON 序列化的 dict（NaN/Inf 也拒）。

## 为什么消息正文全量进 trace（与工具结果的 digest 口径不同）

``agents/runtime.py`` 里工具结果只落 ``result_digest``，理由是结果可能是大数组。
消息不同：它是阶段之间**唯一的**信息通道，M5-4 要核对「消息全部可见、不存在
隐藏状态」。若只落 digest，核对就必须重放全部代码才能读出消息内容 —— 那不叫
核对，叫复算。发布方有责任保证 payload 是**小结**；大数组留在工具结果里，由
call_id 引用。本模块用序列化检查把「不可序列化的东西」挡在发布之前。

## 设计决定（非事实断言）

* **同步、单线程、确定性**：``publish`` 立即投递给当前订阅者；没有异步队列，
  没有后台线程，没有定时器。投递顺序 = 发布顺序 = trace 顺序。
* **msg_id 只由发布顺序决定**（``m0001`` 起）。不含时间戳或随机量 —— 这是
  回放式评测「同输入两跑逐字节相同」的前提（同 ``trace/writer.py`` 的纪律）。
* **未订阅的主题照常入 log**：总线是广播记录，不是 RPC。没有收件人不算错，
  但消息必须留痕。
* **重复订阅直接报错**（同 ``ToolRegistry.register`` 的重名口径）：静默去重会让
  「我以为订上了」与「其实没订」在行为上无法区分。
* **``inbox`` 对未订阅的名字抛 KeyError**：接线错（名字拼错）应该在链路的第一轮

def observation_payload(
    observations: list[Observation],
    *,
    system: str = "GPS",
    adr_sign: str = "adr",
    max_groups: int = 4,

    """把观测批次整理成一条可发布的消息 payload。

    每个卫星组的 ``tool_args`` 就是 ``slip_detect`` 的完整入参（序列行沿用
    ``tests/test_agent_config.py::_slip_args`` 的口径）—— 模型照抄即可调用，
    不存在「把数据改写成工具要的形状」这一步自由发挥。

    * 只接受**单站**批次：混站会让「这批数据属于哪站」在消息里消失；
    * 大序列按 ``max_groups`` 截断（组数与总量都记在 payload 里，截断可见）；
    * 空批次直接报错：发布一条空的观测消息，下游会以为「数据没问题」。
    """
    if not observations:
        raise ValueError("空观测批次：没有可发布的观测消息")
    stations = {ob.station for ob in observations}
    if len(stations) != 1:
        raise ValueError(f"observation_payload 只接受单站批次，收到 {sorted(stations)}")

    pairs = pair_observations(observations)
    signal_1, signal_2 = DEFAULT_PAIR[system]
    sign = meas_cycle_sign(adr_sign)
    groups: list[dict] = []
    for station_id, sys_name, sv in sorted(pairs):
        if sys_name != system:
            continue
        series = pairs[(station_id, sys_name, sv)]
        rows = []

            row: dict = {
                "gps_week": epoch.gps_week,
                "gps_tow_s": epoch.gps_tow_s,
                "quality_flag": epoch.quality_flag,
            }
            for signal in (signal_1, signal_2):
                ob = epoch.get(signal)
                row[signal] = None if ob is None else {
                    "psr_m": ob.psr_m, "adr_cycles": ob.adr_cycles}
            rows.append(row)
        groups.append({
            "key": f"{sys_name} {sv:02d}",
            "n_epochs": len(series),

------------------------------------------------------------------------------

            "tool_args": {
                "series": rows,

def _read_metrics(path: Path) -> list[dict]:
    events: list[dict] = []
    if not path.is_file():
        return events
    for line in path.read_text(encoding="utf-8").splitlines():
        if line.strip():
            events.append(json.loads(line))
    return events


def _token_totals(metrics_events: list[dict]) -> dict[str, dict]:
    totals: dict[str, dict] = {}
    for event in metrics_events:
        if event.get("kind") != "provider_metrics":
            continue
        bucket = totals.setdefault(event.get("agent"), {
            "calls": 0, "prompt_tokens": 0, "completion_tokens": 0})
        usage = event.get("usage") or {}
        bucket["calls"] += 1
        bucket["prompt_tokens"] += int(usage.get("prompt_tokens", 0) or 0)
        bucket["completion_tokens"] += int(usage.get("completion_tokens", 0) or 0)
    return totals


def _token_expectations(scripted: dict[str, list[ChatResponse]]) -> dict[str, dict]:
    expected: dict[str, dict] = {}
    for stage, responses in scripted.items():
        expected[stage] = {
            "calls": len(responses),
            "prompt_tokens": sum(int((r.usage or {}).get("prompt_tokens", 0))
                                 for r in responses),
            "completion_tokens": sum(int((r.usage or {}).get("completion_tokens", 0))
                                     for r in responses),
        }
    return expected

def _write_json(path: Path, obj: dict) -> None:
    # 钉 LF（同 harness/m4_agents_replay.py）：Windows 默认写 CRLF，会让「照
    # REPRODUCE 跑一遍」后工作区里的日志/fixture 与入库版本字节不同，而
    # .gitattributes 的归一化又让 git status 看不出来 —— 沉默的差异会被当成结论。
    path.parent.mkdir(parents=True, exist_ok=True)
    with path.open("w", encoding="utf-8", newline="\n") as fh:
        fh.write(json.dumps(obj, ensure_ascii=False, indent=2, sort_keys=False) + "\n")

def _compare(fixture: dict, log: dict) -> tuple[dict[str, bool], list[str]]:
    """逐阶段与顶层比较；返回 (每阶段是否一致, 问题清单)。"""
    problems: list[str] = []
    expected = {s["stage"]: s for s in fixture.get("stages", [])}
    got = {s["stage"]: s for s in log["stages"]}
    stage_ok: dict[str, bool] = {}
    if set(expected) != set(got):
        problems.append(
            f"阶段集合不一致：fixture {sorted(expected)} / 实跑 {sorted(got)}")

        exp = expected.get(name) or {}
        ok = all(exp.get(key) == got[name].get(key)
                 for key in ("config_hash", "sha256"))
        stage_ok[name] = ok
        if not ok:
            problems.append(
                f"{name} 漂移：config_hash {_short(str(exp.get('config_hash')))}"
                f"→{_short(str(got[name].get('config_hash')))}、"

------------------------------------------------------------------------------
野外台站智能体运行与评估系统 V1.0                                          第 147 页 / 共 169 页
                f"sha256 {_short(str(exp.get('sha256')))}"
                f"→{_short(str(got[name].get('sha256')))}")
    for key in ("checks", "summary", "sample"):

    try:

        print(f"[verify] 读不了日志 {path}：{exc}", file=sys.stderr)
        return 2
    if log.get("kind") != LOG_KIND:
        print(f"[verify] {path} 不是 {LOG_KIND}（kind={log.get('kind')!r}）",
              file=sys.stderr)
        return 2

    problems: list[str] = []
    stages = log.get("stages") or []

            continue

"""回放式评测器：合成 fixture -> 智能体单轮 -> 归因结论 + trace -> SHA256 对拍。

用法::

    python harness/replay.py --fixture harness/fixtures/replay_20261013.json
    python harness/replay.py --fixture <path> --log-dir <dir>  # 日志落别处（测试用）
    python harness/replay.py --fixture <path> --bless          # 重建期望

退出码（tests/test_replay.py 两条路径钉死）：

- ``0``  归因输出与 fixture 期望一致（或 ``--bless`` 成功）
- ``1``  不一致：``expected_sha256`` 或 ``config_hash`` 任一漂移
- ``2``  用法 / IO 错误（缺 ``--fixture``、fixture 不存在且未 ``--bless``、
  JSON 坏、缺字段、kind 不对）

链路（每一步只由种子与代码决定，不碰真实数据、不调真实模型）::

    synthetic.default_cfg(seed=fixture.seed)
      -> synthetic.generate(...)                    # 合成观测 + 生效配置
      -> derive_user_input(...)                     # 提问从观测派生（数据驱动）
      -> runtime.run(agent, user_input, trace)      # FakeProvider 单轮
      -> sha256({"conclusion": .., "trace": [..]})  # 归因结论 + 整个 trace
      -> 与 fixture.expected_sha256 对拍，写 logs/replay_<seed>.json

**哈希为什么可以直接对整个 trace 算（不存在"去时间戳"这一步）**：
trace 设计上就不含时间戳与随机量（`agents/trace/writer.py` 顶部），测试上被钉死
（`tests/test_runtime.py::TestRun::test_two_identical_runs_produce_identical_trace`
要求同输入两次运行 trace 逐字节相同）。本机实测（2026-10-13，两次**独立进程**
跑同一 fixture）::

    run1 sha256: 775435d08a5da1cc1539326fbef750c3f635f1098951d98d484356a124bbd14a
    run2 sha256: 775435d08a5da1cc1539326fbef750c3f635f1098951d98d484356a124bbd14a

两次完全一致，故对 trace 原文整体哈希即可。更早的设想「先去掉 trace 里的
时间戳再哈希」建立在一个后来被证伪的假设上 —— trace 根本没有时间戳可去。

**M0 不接真实模型**：provider 固定 ``FakeProvider``，响应是本文件的
``RESPONSES`` 常量。M3 换真实模型时只需换 provider，对拍结构不变，但结论
必然不同 —— 那时基线要 ``--bless`` 有意重建，这是设计内的事，不是漂移。

    """返回 [(项, 期望, 实测, 是否一致), ...]。缺字段由调用方先行拦截（退出码 2）。"""
